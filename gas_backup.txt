// ========== DO GET (UNTUK MEMBACA DATA) ==========
function doGet(e) {
  const action = e?.parameter?.action || '';
  const callback = e?.parameter?.callback || '';
  
  let result;
  
  if (action === 'getAllData') {
    result = getAllDataResponse();
  } else {
    result = {
      status: 'ok',
      message: 'API berjalan',
      produk: getProdukDariSheet(),
      timestamp: new Date().toISOString()
    };
  }
  
  // JSONP untuk CORS
  if (callback) {
    return ContentService.createTextOutput(`${callback}(${JSON.stringify(result)})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// ========== DO POST (UNTUK MENYIMPAN DATA) ==========
function doPost(e) {
  const output = ContentService.createTextOutput();
  output.setMimeType(ContentService.MimeType.JSON);
  
  try {
    let data = {};
    
    if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else if (e.parameter && e.parameter.data) {
      data = JSON.parse(e.parameter.data);
    } else if (e.parameter) {
      data = e.parameter;
    }
    
    const action = data.action;
    
    // Catat aktivitas ke log
    logActivity('POST_REQUEST', action, JSON.stringify(data).substring(0, 200));
    
    if (action === 'saveTransaction') {
      const result = saveTransaction(data.transaction);
      output.setContent(JSON.stringify(result));
    } else if (action === 'payDebt') {
      const result = payDebt(data.nama);
      output.setContent(JSON.stringify(result));
    } else if (action === 'updateSettings') {
      const result = updateSettings(data.settings);
      output.setContent(JSON.stringify(result));
    } else if (action === 'saveProduct') {
      const result = saveProduct(data.product);
      output.setContent(JSON.stringify(result));
    } else if (action === 'savePetugas') {
      const result = savePetugas(data.petugas);
      output.setContent(JSON.stringify(result));
    } else if (action === 'deletePetugas') {
      const result = deletePetugas(data.id);
      output.setContent(JSON.stringify(result));
    } else {
      output.setContent(JSON.stringify({ status: 'ok', message: 'API ready' }));
    }
    
  } catch(error) {
    logActivity('ERROR', 'doPost', error.toString());
    output.setContent(JSON.stringify({ status: 'error', message: error.toString() }));
  }
  
  return output;
}

// ========== BACA PRODUK DARI SHEET ==========

      const row = rows[i];
      if (row[0] && row[0].toString().trim() !== '') {
        produk.push({
          id: row[0].toString(),
          nama: row[1] || 'Produk',

        });

    }
    
    return produk;
  } catch(e) {
    logActivity('ERROR', 'getProdukDariSheet', e.toString());

}

// ========== BACA SEMUA DATA UNTUK DASHBOARD ==========
function getAllDataResponse() {
  const produk = getProdukDariSheet();
  let transaksi = [];
  let pelanggan = [];
  
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const trxSheet = ss.getSheetByName('transaksi');
    
    if (trxSheet) {
      const trxRows = trxSheet.getDataRange().getValues();
      
      // Header: 0=id,1=tanggal,2=waktu,3=petugas_id,4=petugas_nama,
      // 5=nama_pelanggan,6=wa,7=items,8=ongkir,9=subtotal,10=total,
      // 11=bayar,12=status,13=catatan,14=hutang_lunas,15=total_galon
      
      for (let i = 1; i < trxRows.length; i++) {
        const row = trxRows[i];
        if (row[0] && row[0].toString().trim() !== '') {
          
          let items = [];
          try {
            if (row[7] && typeof row[7] === 'string') {
              items = JSON.parse(row[7]);
            } else if (row[7]) {
              items = row[7];
            }
          } catch(e) {
            items = [];
          }
          
          // Ambil tanggal dalam format YYYY-MM-DD
          let tanggal = '';
          if (row[1]) {
            tanggal = row[1].toString();
            if (tanggal.includes('T')) {
              tanggal = tanggal.split('T')[0];
            }
          }
          
          transaksi.push({
            id: row[0].toString(),
            tanggal: tanggal,
            waktu: row[2] ? row[2].toString() : '',
            petugas_id: row[3] ? row[3].toString() : '',
            petugas_nama: row[4] ? row[4].toString() : '',
            nama: row[5] ? row[5].toString() : 'Umum',
            wa: row[6] ? row[6].toString() : '',
            items: items,
            ongkir: Number(row[8]) || 0,
            subtotal: Number(row[9]) || 0,
            total: Number(row[10]) || 0,
            bayar: row[11] ? row[11].toString() : 'tunai',
            status: row[12] ? row[12].toString() : 'lunas',
            catatan: row[13] ? row[13].toString() : '',
            hutangLunas: row[14] === 'TRUE' || row[14] === true,
            totalGalon: Number(row[15]) || 0
          });
        }
      }
    }
    
    // Hitung pelanggan unik
    const pelangganMap = new Map();
    transaksi.forEach(t => {
      const namaPelanggan = t.nama || 'Umum';
      if (!pelangganMap.has(namaPelanggan)) {
        pelangganMap.set(namaPelanggan, {
          nama: namaPelanggan,
          wa: t.wa || '',
          total_order: 0,
          hutang: 0
        });
      }
      const p = pelangganMap.get(namaPelanggan);
      p.total_order += t.total;
      if (t.status === 'hutang' && !t.hutangLunas) {
        p.hutang += t.total;
      }
    });
    pelanggan = Array.from(pelangganMap.values());
    
    logActivity('READ_DATA', 'getAllDataResponse', 'Berhasil baca ' + transaksi.length + ' transaksi');
    
  } catch(e) {
    logActivity('ERROR', 'getAllDataResponse', e.toString());
  }
  
  // Baca settings dari sheet settings
  let settings = { nama_depot: 'EMBUN LAWU', alamat: 'Blumbang, Tawangmangu' };
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const settingsSheet = ss.getSheetByName('settings');
    if (settingsSheet) {
      const data = settingsSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        const key = data[i][0];
        const value = data[i][1];
        if (key && value) {
          settings[key] = value;
        }
      }
    }
  } catch(e) {}
  
  return {
    status: 'ok',
    data: {
      produk: produk,
      settings: settings,
      pelanggan: pelanggan,
      transaksi: transaksi,
      petugas: getPetugasDariSheet()
    }
  };
}

// ========== SIMPAN TRANSAKSI ==========
function saveTransaction(trx) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let sheet = ss.getSheetByName('transaksi');
    
    // Buat sheet dengan header yang benar jika belum ada
    if (!sheet) {
      sheet = ss.insertSheet('transaksi');
      sheet.appendRow([
        'id', 'tanggal', 'waktu', 'petugas_id', 'petugas_nama', 
        'nama_pelanggan', 'wa', 'items', 'ongkir', 'subtotal', 
        'total', 'bayar', 'status', 'catatan', 'hutang_lunas', 'total_galon'
      ]);
      logActivity('CREATE_SHEET', 'transaksi', 'Membuat sheet transaksi baru');
    } else {
      // Perbaiki header jika masih salah
      const headerRow = sheet.getRange(1, 1, 1, 16).getValues()[0];
      if (headerRow[4] !== 'petugas_nama' || headerRow[5] !== 'nama_pelanggan') {
        sheet.getRange(1, 1, 1, 16).setValues([[
          'id', 'tanggal', 'waktu', 'petugas_id', 'petugas_nama', 
          'nama_pelanggan', 'wa', 'items', 'ongkir', 'subtotal', 
          'total', 'bayar', 'status', 'catatan', 'hutang_lunas', 'total_galon'
        ]]);
        logActivity('FIX_HEADER', 'transaksi', 'Memperbaiki header sheet');
      }
    }
    
    // ========== HITUNG TOTAL GALON ==========
    let totalGalon = 0;
    
    if (trx.items && Array.isArray(trx.items)) {
      for (let i = 0; i < trx.items.length; i++) {
        const item = trx.items[i];
        if (item.satuan && item.satuan.toLowerCase() === 'galon') {
          const qty = item.qty || 0;
          totalGalon += qty;
        }
      }
    }
    
    // Format tanggal
    let tanggal = trx.tanggal;
    if (tanggal && tanggal.includes('T')) {
      tanggal = tanggal.split('T')[0];
    }
    
    // Data yang akan disimpan
    const rowData = [
      trx.id || 'TRX_' + Date.now(),
      tanggal || new Date().toISOString().slice(0, 10),
      trx.waktu || new Date().toISOString(),
      trx.petugas_id || 'P001',
      trx.petugas_nama || 'KHOLID',
      trx.nama || 'Umum',
      trx.wa || '',
      JSON.stringify(trx.items || []),
      trx.ongkir || 0,
      trx.subtotal || trx.total || 0,
      trx.total || 0,
      trx.bayar || 'tunai',
      trx.status || (trx.bayar === 'hutang' ? 'hutang' : 'lunas'),
      trx.catatan || '',
      trx.hutangLunas ? 'TRUE' : 'FALSE',
      totalGalon
    ];
    
    sheet.appendRow(rowData);
    
    // Catat ke log aktivitas
    logActivity('SAVE_TRANSACTION', trx.id, 
      'Pelanggan: ' + (trx.nama || 'Umum') + 
      ', Total: Rp' + (trx.total || 0) + 
      ', Galon: ' + totalGalon);
    
    // Update backup_meta (terakhir transaksi)
    updateBackupMeta('last_transaction', new Date().toISOString());
    
    return {
      status: 'ok',
      message: 'Transaksi tersimpan',
      id: trx.id,
      totalGalon: totalGalon
    };
    
  } catch(e) {
    logActivity('ERROR', 'saveTransaction', e.toString());
    return {
      status: 'error',
      message: e.toString()
    };
  }
}

// Fungsi untuk memperbaiki data transaksi yang sudah ada
function fixExistingGalonData() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheetByName('transaksi');
  
  if (!sheet) {
    return { status: 'error', message: 'Sheet transaksi tidak ditemukan' };
  }
  
  // Perbaiki header
  sheet.getRange(1, 1, 1, 16).setValues([[
    'id', 'tanggal', 'waktu', 'petugas_id', 'petugas_nama', 
    'nama_pelanggan', 'wa', 'items', 'ongkir', 'subtotal', 
    'total', 'bayar', 'status', 'catatan', 'hutang_lunas', 'total_galon'
  ]]);
  
  const data = sheet.getDataRange().getValues();
  let fixedCount = 0;
  let totalGalonAll = 0;
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    let totalGalon = 0;
    
    if (row[7]) {
      try {
        let items = [];
        if (typeof row[7] === 'string') {
          items = JSON.parse(row[7]);
        } else if (Array.isArray(row[7])) {
          items = row[7];
        }
        
        for (let j = 0; j < items.length; j++) {
          const item = items[j];
          if (item.satuan && item.satuan.toLowerCase() === 'galon') {
            totalGalon += item.qty || 0;
          }
        }
        
        if (totalGalon > 0 || row[15] != totalGalon) {
          sheet.getRange(i + 1, 16).setValue(totalGalon);
          fixedCount++;
          totalGalonAll += totalGalon;
        }
      } catch(e) {}
    }
  }
  
  logActivity('FIX_DATA', 'fixExistingGalonData', 'Memperbaiki ' + fixedCount + ' transaksi, total ' + totalGalonAll + ' galon');
  
  return { 
    status: 'ok', 
    fixed: fixedCount,
    totalGalon: totalGalonAll
  };
}

// Fungsi untuk cek isi sheet
function checkSheetData() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheets = ss.getSheets();
  const result = {
    sheets: sheets.map(s => s.getName()),
    transaksi: { count: 0, lastRow: null },
    produk: { count: 0 }
  };
  
  const trxSheet = ss.getSheetByName('transaksi');
  if (trxSheet) {
    const data = trxSheet.getDataRange().getValues();
    result.transaksi.count = data.length - 1;
    if (data.length > 1) {
      result.transaksi.lastRow = data[data.length - 1];
    }
  }
  
  const produkSheet = ss.getSheetByName('produk');
  if (produkSheet) {
    const data = produkSheet.getDataRange().getValues();
    result.produk.count = data.length - 1;
  }
  
  return result;
}

// Backup otomatis (bisa dijadikan trigger waktu)
function createBackup() {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const backupName = 'backup_' + new Date().toISOString().slice(0, 19).replace(/:/g, '-');
    const backup = SpreadsheetApp.create(backupName);
    
    // Copy semua sheet
    const sheets = ss.getSheets();
    sheets.forEach(sheet => {
      const sheetName = sheet.getName();
      const newSheet = backup.insertSheet(sheetName);
      const data = sheet.getDataRange().getValues();
      newSheet.getRange(1, 1, data.length, data[0].length).setValues(data);
    });
    
    // Update meta backup
    updateBackupMeta('last_backup', new Date().toISOString());
    updateBackupMeta('last_backup_file', backup.getUrl());
    
    logActivity('BACKUP', 'createBackup', 'Backup created: ' + backup.getUrl());
    
    return { status: 'ok', url: backup.getUrl() };
  } catch(e) {
    logActivity('ERROR', 'createBackup', e.toString());
    return { status: 'error', message: e.toString() };
  }
}

// ========== PELUNASAN HUTANG PELANGGAN ==========
function payDebt(nama) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const sheet = ss.getSheetByName('transaksi');
    if (!sheet) return { status: 'error', message: 'Sheet transaksi tidak ditemukan' };
    
    const rows = sheet.getDataRange().getValues();
    let updatedCount = 0;
    
    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      // Kolom 5 = nama_pelanggan, Kolom 12 = status, Kolom 14 = hutang_lunas
      if (row[5] === nama && row[12] === 'hutang' && (row[14] === 'FALSE' || row[14] === false || !row[14])) {
        // Update kolom status ke 'lunas' (Kolom 13, Indeks ke-13)
        sheet.getRange(i + 1, 13).setValue('lunas');
        // Update kolom hutang_lunas ke TRUE (Kolom 15, Indeks ke-15)
        sheet.getRange(i + 1, 15).setValue('TRUE');
        updatedCount++;
      }
    }
    
    logActivity('PAY_DEBT', nama, 'Berhasil melunasi ' + updatedCount + ' transaksi hutang');
    return { status: 'ok', message: 'Sukses melunasi ' + updatedCount + ' transaksi hutang', count: updatedCount };
  } catch(e) {
    logActivity('ERROR', 'payDebt', e.toString());
    return { status: 'error', message: e.toString() };
  }
}

// ========== UPDATE SETTINGS ==========
function updateSettings(settings) {
  try {

      const data = sheet.getDataRange().getValues();

          sheet.getRange(i + 1, 3).setValue(new Date().toISOString());

  } catch(e) {
    logActivity('ERROR', 'updateSettings', e.toString());

  }

// ========== SAVE / UPDATE PRODUCT ==========
function saveProduct(product) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let sheet = ss.getSheetByName('produk');
    if (!sheet) {
      sheet = ss.insertSheet('produk');
      sheet.appendRow(['id', 'nama', 'icon', 'satuan', 'harga_beli', 'harga', 'stok', 'is_active']);
    }
    
    const data = sheet.getDataRange().getValues();
    let foundIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (data[i][0].toString() === product.id.toString()) {
        foundIndex = i;
        break;
      }
    }
    
    const rowData = [
      product.id,
      product.nama,
      product.icon || '💧',
      product.satuan || 'galon',
      Number(product.harga_beli) || 0,
      Number(product.harga) || 0,
      Number(product.stok) || 0,
      product.is_active ? true : false
    ];
    
    if (foundIndex !== -1) {
      sheet.getRange(foundIndex + 1, 1, 1, 8).setValues([rowData]);
    } else {
      sheet.appendRow(rowData);
    }
    
    logActivity('SAVE_PRODUCT', product.id, 'Nama: ' + product.nama + ', Harga: ' + product.harga);
    return { status: 'ok', message: 'Produk berhasil disimpan' };
  } catch(e) {
    logActivity('ERROR', 'saveProduct', e.toString());
    return { status: 'error', message: e.toString() };
  }
}

// ========== GET STAFF FROM SHEET ==========
function getPetugasDariSheet() {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const sheet = ss.getSheetByName('petugas');
    if (!sheet) {
      return [{ id: 'P001', nama: 'KHOLID', jabatan: 'Owner' }];
    }
    
    const rows = sheet.getDataRange().getValues();
    const headers = rows[0];
    const is6Cols = headers.length >= 4 && headers[2] === 'is_active';
    
    const petugas = [];
    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      if (row[0] && row[0].toString().trim() !== '') {
        const jab = is6Cols ? (row[3] || 'Kasir') : (row[2] || 'Kasir');
        const isActive = is6Cols ? (row[2] === true || row[2] === 'TRUE' || row[2] === 'True') : true;
        
        if (isActive) {
          petugas.push({
            id: row[0].toString(),
            nama: row[1] || 'Staff',
            jabatan: jab.toString()
          });
        }
      }
    }
    return petugas.length > 0 ? petugas : [{ id: 'P001', nama: 'KHOLID', jabatan: 'Owner' }];
  } catch(e) {
    return [{ id: 'P001', nama: 'KHOLID', jabatan: 'Owner' }];
  }
}

// ========== SAVE STAFF ==========
function savePetugas(p) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let sheet = ss.getSheetByName('petugas');
    if (!sheet) {
      sheet = ss.insertSheet('petugas');
      sheet.appendRow(['id', 'nama', 'is_active', 'jabatan', 'created_at', 'last_login']);
      sheet.appendRow(['P001', 'KHOLID', 'TRUE', 'Owner', new Date().toISOString().split('T')[0], '']);
    }
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const is6Cols = headers.length >= 4 && headers[2] === 'is_active';
    
    let foundIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (data[i][0].toString() === p.id.toString()) {
        foundIndex = i;
        break;
      }
    }
    
    let rowData;
    if (is6Cols) {
      const todayStr = new Date().toISOString().split('T')[0];
      rowData = [
        p.id,
        p.nama,
        true,
        p.jabatan || 'Kasir',
        foundIndex !== -1 ? (data[foundIndex][4] || todayStr) : todayStr,
        foundIndex !== -1 ? (data[foundIndex][5] || '') : ''
      ];
      if (foundIndex !== -1) {
        sheet.getRange(foundIndex + 1, 1, 1, 6).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
    } else {
      rowData = [p.id, p.nama, p.jabatan || 'Kasir'];
      if (foundIndex !== -1) {
        sheet.getRange(foundIndex + 1, 1, 1, 3).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
    }
    
    logActivity('SAVE_PETUGAS', p.id, 'Nama: ' + p.nama + ', Jabatan: ' + p.jabatan);
    return { status: 'ok', message: 'Petugas berhasil disimpan' };
  } catch(e) {
    logActivity('ERROR', 'savePetugas', e.toString());
    return { status: 'error', message: e.toString() };
  }
}

// ========== DELETE STAFF ==========
function deletePetugas(id) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const sheet = ss.getSheetByName('petugas');
    if (!sheet) return { status: 'error', message: 'Sheet petugas tidak ditemukan' };
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const is6Cols = headers.length >= 4 && headers[2] === 'is_active';
    
    for (let i = 1; i < data.length; i++) {
      if (data[i][0].toString() === id.toString()) {
        if (is6Cols) {
          sheet.getRange(i + 1, 3).setValue(false);
          logActivity('DELETE_PETUGAS_SOFT', id, 'Berhasil menonaktifkan petugas');
          return { status: 'ok', message: 'Petugas berhasil dinonaktifkan' };
        } else {
          sheet.deleteRow(i + 1);
          logActivity('DELETE_PETUGAS_HARD', id, 'Berhasil menghapus petugas');
          return { status: 'ok', message: 'Petugas berhasil dihapus' };
        }
      }
    }
    return { status: 'error', message: 'Petugas tidak ditemukan' };
  } catch(e) {
    logActivity('ERROR', 'deletePetugas', e.toString());
    return { status: 'error', message: e.toString() };
  }
}
